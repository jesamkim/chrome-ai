# GitHub 저장소 생성 후 실행할 명령어들

# 1. 현재 디렉토리 확인
cd /Workshop/chrome-ai

# 2. Git 상태 확인
git status

# 3. 원격 저장소 추가 (이미 추가되어 있음)
git remote -v

# 4. GitHub에 푸시
git push -u origin main

# 5. 푸시 성공 확인
echo "✅ 푸시 완료! 저장소 URL: https://github.com/jesamkim/chrome-bedrock-chat"
